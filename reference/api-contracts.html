Draft API Contracts

// Inference
POST /api/infer
{ modelId: string; features: Record<string, number>; }
→ { prediction: string; confidence: number; probabilities: Record<string, number>; timestamp: string }

// Training
POST /api/train
{ datasetId: string; architecture: string; hyperparams: Record<string,string|number>; }
→ { jobId: string }

// Batch
POST /api/batch
{ modelId: string; fileUploadId: string }
→ { jobId: string }
GET /api/batch/:jobId
→ { status: 'pending'|'running'|'complete'|'error'; resultsUrl?: string }

// Export
GET /api/models/:id/export
→ { downloadUrl: string; format: 'json'|'onnx'|'bundle' }

Error Envelope:

{"error": {"code": "VALIDATION_ERROR", "message": "Missing field: features"}}
